cmake_minimum_required(VERSION 3.16)
project(visual_graph_tool VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)
    add_compile_options(/W4 /permissive- /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()

add_library(graph_tool src/config.cpp src/data.cpp)
target_include_directories(graph_tool PUBLIC src)
if(WIN32)
    target_link_libraries(graph_tool PRIVATE winhttp)
endif()
add_executable(visual_graph_tool src/main.cpp)
target_link_libraries(visual_graph_tool PRIVATE graph_tool)
if(MINGW)
    target_link_options(visual_graph_tool PRIVATE -municode)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(unit_tests tests/unit_tests.cpp)
    target_link_libraries(unit_tests PRIVATE graph_tool)
    add_test(NAME unit_tests COMMAND unit_tests)
    set_tests_properties(unit_tests PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
    if(WIN32)
        add_executable(cli_tests tests/test_cli.cpp)
        target_link_libraries(cli_tests PRIVATE graph_tool ws2_32)
        if(MINGW)
            target_link_options(cli_tests PRIVATE -municode)
        endif()
        add_test(NAME cli_tests COMMAND cli_tests
            "$<TARGET_FILE:visual_graph_tool>" "${PROJECT_SOURCE_DIR}")
        set_tests_properties(cli_tests PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" TIMEOUT 90)
    endif()
endif()
